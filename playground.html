// Golang: High-throughput concurrent SKU Lookup with Redis Cache-Aside
func (s *InventoryService) GetSKU(ctx context.Context, barcode string) (*Item, error) {
    cacheKey := fmt.Sprintf("sku:item:%s", barcode)
    
    // 1. O(1) Fast Path: Check in-memory Redis cluster
    if val, err := s.redis.Get(ctx, cacheKey).Bytes(); err == nil {
        var item Item
        if err := json.Unmarshal(val, &item); err == nil {
            telemetry.RecordCacheHit("redis") // ~2.4ms latency
            return &item, nil
        }
    }

    // 2. Slow Path: Fallback to Amazon RDS MySQL with Read Replica
    item, err := s.dbRepo.FindByBarcode(ctx, barcode) // ~130ms latency
    if err != nil {
        return nil, err
    }

    // 3. Populate Redis asynchronously with TTL
    go func(itm *Item) {
        data, _ := json.Marshal(itm)
        s.redis.Set(context.Background(), cacheKey, data, 5*time.Minute)
    }(item)

    return item, nil
}
<?php
namespace App\Services\AYOKasir;

use Illuminate\Support\Facades\Cache;
use App\Models\InventoryItem;

class SkuLookupService
{
    /**
     * Look up inventory item with resilient Redis Cache-Aside pattern.
     */
    public function getByBarcode(string $barcode): InventoryItem
    {
        $cacheKey = "sku:item:{$barcode}";

        // Atomic remember: Cache HIT returns in ~2.5ms; MISS executes DB query
        return Cache::store('redis')->remember($cacheKey, now()->addMinutes(10), function () use ($barcode) {
            \Log::channel('telemetry')->info("CACHE MISS: Querying RDS MySQL disk for barcode: {$barcode}");
            
            return InventoryItem::where('barcode', $barcode)
                ->select(['id', 'sku_code', 'name', 'price', 'stock_quantity'])
                ->firstOrFail();
        });
    }
}
